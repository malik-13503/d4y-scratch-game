### ✅ Example Code Snippet to Insert

This is the updated `handleSpin()` function:

```ts
const handleSpin = async () => {
  if (isSpinning || disabled) return;

  setIsSpinning(true);
  setResult(null);
  setShowResultModal(false);

  try {
    const spinResult = await onSpin();
    const segmentIndex = wheelNumbers.findIndex(num => num === spinResult);
    const totalSegments = segmentColors.length;

    if (segmentIndex === -1) {
      console.warn("Spin result not in wheel numbers");
      setResult(spinResult);
      setIsFreePlay(spinResult >= freePlayStart);
      setAmountCharged(spinResult >= freePlayStart ? 0 : spinResult);
      setIsSpinning(false);
      setShowResultModal(true);
      return;
    }

    const segmentAngle = 360 / totalSegments;
    const targetAngle = segmentIndex * segmentAngle + segmentAngle / 2;
    const fullRotations = 4 * 360;
    const finalRotation = fullRotations + (360 - targetAngle);

    setRotation(rotation + finalRotation);

    setTimeout(() => {
      setResult(spinResult);
      const isFree = spinResult >= freePlayStart;
      setIsFreePlay(isFree);
      setAmountCharged(isFree ? 0 : spinResult);
      setIsSpinning(false);
      setShowResultModal(true);
    }, 3000);
  } catch (err) {
    console.error("Spin error:", err);
    setIsSpinning(false);
    setResult(Math.floor(Math.random() * totalNumbers) + 1);
    setIsFreePlay(false);
    setAmountCharged(50);
    setShowResultModal(true);
  }
};
```
